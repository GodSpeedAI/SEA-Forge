fn sha(data: &[u8]) -> String {
    format!("{:x}", Sha256::digest(data))
}

fn obj(v: Value) -> Map<String, Value> {
    v.as_object().unwrap().clone()
}

fn tmp(tag: &str) -> PathBuf {
    let d = std::env::temp_dir().join(format!("t01-verifier-{}-{}", tag, std::process::id()));
    let _ = std::fs::remove_dir_all(&d);
    std::fs::create_dir_all(&d).unwrap();
    d
}

fn report(name: &str, result: &str, detail: &str) {
    println!("VERIFIER|{name}|{result}|{detail}");
}

// --- V-A1: causal continuity beyond one hop ---------------------------------

/// Two agreeing parents from DIFFERENT work cycles is the conflict class;
/// here: two parents with CONFLICTING work identities must be fatal.
#[test]
fn va1a_multiple_parents_with_conflicting_work_identities_are_fatal() {
    let h = sha(b"model");
    let p1 = make_event("ContextPacketCreated", obj(json!({"work_request_id": "wr-A"})), &h);
    let p2 = make_event("ProofCompleted", obj(json!({"work_request_id": "wr-B"})), &h);
    match derive_event(&[&p1, &p2], "RouteSelected", obj(json!({})), &h, None) {
        Err(swe_seed_core::federation::DeriveError::CorrelationMismatch { .. }) => {
            report("V-A1a", "CONTRACT-HELD", "parent-vs-parent conflict is CorrelationMismatch");
        }
        other => panic!("conflicting parents accepted: {other:?}"),
    }
}

/// Three-generation chain: correlation survives two derivations unchanged;
/// grandchild identifies its direct parent (transitive chain inspectable).
#[test]
fn va1b_three_generation_derivation_preserves_correlation() {
    let h = sha(b"model");
    let g0 = make_event("WorkRequested", obj(json!({"work_request_id": "wr-Z"})), &h);
    let g1 = derive_event(&[&g0], "ContextRequired", obj(json!({"q": 1})), &h, None).unwrap();
    let g2 = derive_event(&[&g1], "RouteSelected", obj(json!({})), &h, None).unwrap();
    assert_eq!(g1.work_request_id(), Some("wr-Z"));
    assert_eq!(g2.work_request_id(), Some("wr-Z"));
    assert_eq!(g2.causal_parents(), vec![g1.event_id.as_str()]);
    assert_eq!(g1.causal_parents(), vec![g0.event_id.as_str()]);
    // Attempted cross-cycle adoption at gen 3 is fatal.
    match derive_event(&[&g2], "ProofCompleted", obj(json!({})), &h, Some("wr-OTHER")) {
        Err(swe_seed_core::federation::DeriveError::CorrelationMismatch { .. }) => {
            report("V-A1b", "CONTRACT-HELD", "correlation stable x3 generations; late cross-wire fatal");
        }
        other => panic!("generation-3 cross-wire accepted: {other:?}"),
    }
}

// --- V-A2: construction-side bypass around VerifiedDomainIdentity ------------

/// derive_event takes a raw &str hash: a child carrying the FORBIDDEN
/// namespace-derived pseudo-hash can be CONSTRUCTED through the T01
/// derivation surface. Probe whether the validation boundary quarantines it.
#[test]
fn va2_derivation_surface_stamps_fallback_pseudo_hash() {
    let h = sha(b"model");
    let parent = make_event_verified(
        "WorkRequested",
        obj(json!({"work_request_id": "wr-K"})),
        &verify_declared_hash(&h).unwrap(),
    );
    let pseudo = fallback_hash();
    let child = derive_event(
        &[&parent],
        "ContextRequired",
        obj(json!({"q": 1})),
        &pseudo, // attacker-supplied pseudo-hash through ungated param
        None,
    );
    match child {
        Ok(c) => {
            assert_eq!(c.domain_model_hash(), Some(pseudo.as_str()));
            assert!(!c.causal_parents().is_empty());
            // Boundary adjudication:
            let v = validate_envelope(&c, &h);
            let conf = check_conformance(&c);
            let rejected = matches!(&v, Err(ConsumeError::Identity(IdentityError::PlaceholderIdentity { .. })));
            report(
                "V-A2",
                if rejected && !conf.is_conformant() { "QUARANTINED-AT-BOUNDARY (construction-side gap recorded)" } else { "FALSIFIER" },
                &format!("constructed_ok=true validate={v:?} conformant={} pseudo={pseudo}", conf.is_conformant()),
            );
            assert!(rejected, "boundary must reject pseudo-hash child");
            assert!(!conf.is_conformant(), "conformance must flag pseudo-hash");
        }
        Err(e) => report("V-A2", "CONTRACT-HELD", &format!("construction refused: {e}")),
    }
}

// --- V-A3: identity bound to fabricated manifest, no artifact bytes ----------

/// strict_resolve trusts meta.sea_file_hash declared by ANY readable manifest;
/// from_resolved then blesses it. Probe: VerifiedDomainIdentity for a hash
/// whose "model bytes" exist NOWHERE on disk.
#[test]
fn va3_fabricated_manifest_yields_verified_identity_without_artifact() {
    let dir = tmp("va3");
    let phantom = sha(b"attacker-bytes-never-written-as-any-artifact");
    let mdir = dir.join("docs/specs/domains/agentic_capability_loop");
    std::fs::create_dir_all(&mdir).unwrap();
    let manifest = mdir.join("agentic_capability_loop.manifest.json");
    std::fs::write(
        &manifest,
        json!({"meta": {"sea_file_hash": phantom}}).to_string(),
    )
    .unwrap();
    let root = dir.to_string_lossy().into_owned();
    let resolved = strict_resolve(Some(&root), None).expect("strict_resolve must see manifest");
    assert_eq!(resolved.source, HashSource::SeaRoot);
    let identity = VerifiedDomainIdentity::from_resolved(&resolved)
        .expect("from_resolved blessed a phantom hash");
    assert_eq!(identity.as_str(), phantom);
    // Full composed gate accepts an envelope stamped with the phantom identity
    // when the caller's expected-hash resolution shares the same substrate.
    let env = make_event_verified("GovernedWorkRequest", obj(json!({"intent": "x"})), &identity);
    validate_envelope(&env, identity.as_str()).expect("phantom identity passed full gate");
    report(
        "V-A3",
        "SUBSTRATE-TRUST-LIMIT-RECORDED",
        "manifest-declared hash blessed without artifact-byte binding; placeholder defense still held below",
    );
}

// --- V-A4: exclusive event authority: case/format/alias + path divergence ----

#[test]
fn va4_producer_forge_format_attacks_fail_closed() {
    let h = sha(b"model");
    let mk = |src: &str| {
        let mut e = make_event("EvidenceRecorded", obj(json!({})), &h);
        e.source_agent = src.to_string();
        e
    };
    let attacks = [
        ("RealityTrace", "mixed case"),
        ("REALITYTRACE", "upper case"),
        (" realitytrace", "leading space"),
        ("realitytrace ", "trailing space"),
        ("reality_trace", "underscore variant"),
        ("sxr ", "alias plus space"),
        ("Reality-Tracer", "near-miss spelling"),
    ];
    for (src, why) in attacks {
        let e = mk(src);
        assert!(
            validate_envelope(&e, &h).is_err(),
            "case/format forge '{src}' ({why}) must fail closed"
        );
    }
    // Event-type aliases / formats:
    for ev in ["evidence_recorded", "Evidencerecorded", "EVIDENCERECORDED", "Evidence Recorded"] {
        let mut e = make_event(ev, obj(json!({})), &h);
        e.source_agent = "realitytrace".into();
        assert!(
            validate_envelope(&e, &h).is_err(),
            "event alias '{ev}' must fail closed"
        );
    }
    // Alias table sanity: documented aliases map onto the SAME authority.
    assert_eq!(canonical_agent("sxr"), canonical_agent("realitytrace"));
    report("V-A4", "CONTRACT-HELD", "7 agent-format + 4 event-alias forgeries all failed closed");
}

// --- V-A5: durable idempotency beyond the happy path -------------------------

/// Altered envelope carrying the SAME stable identity after restart:
/// the rewrite attempt must not produce a second admission.
#[test]
fn va5a_altered_payload_same_identity_after_restart_is_suppressed() {
    let dir = tmp("va5a");
    let path = dir.join("ledger.jsonl");
    let h = sha(b"model");
    let original = make_event("ProofCompleted", obj(json!({"result": "failed"})), &h);
    let key = original.idempotency_key.clone().unwrap();

    let mut p1 = IdempotencyLedger::open(&path).unwrap();
    assert_eq!(p1.admit(&key).unwrap(), Admission::First);
    drop(p1);

    // Attacker/buggy rewriter changes the FACT but keeps the identity.
    let mut rewritten = original.clone();
    rewritten.payload["result"] = json!("passed");
    assert_eq!(
        rewritten.idempotency_key.as_deref(),
        Some(key.as_str()),
        "rewrite keeps stable identity"
    );

    let mut p2 = IdempotencyLedger::open(&path).unwrap(); // restart
    assert_eq!(p2.admit(&key).unwrap(), Admission::Duplicate);
    assert_eq!(p2.len(), 1);
    // Genuinely different content remains a distinct event:
    let other = make_event("ProofCompleted", obj(json!({"result": "passed"})), &h);
    assert_ne!(other.idempotency_key.as_deref(), Some(key.as_str()));
    assert_eq!(p2.admit(other.idempotency_key.as_deref().unwrap()).unwrap(), Admission::First);
    report("V-A5a", "CONTRACT-HELD", "same-identity rewrite suppressed after restart; distinct content distinct key");
}

/// Concurrent racing admitters over ONE ledger file (two-process model):
/// count how many observe First. Duplicate lines converge on reload.
#[test]
fn va5b_concurrent_admission_race_over_shared_file() {
    let dir = tmp("va5b");
    let path = Arc::new(dir.join("ledger.jsonl"));
    let barrier = Arc::new(Barrier::new(2));
    let key = idempotency_key("EvidenceRecorded", &json!({"n": 1}), Some("wr-race"));
    let mut handles = Vec::new();
    for _ in 0..2 {
        let p = path.clone();
        let b = barrier.clone();
        let k = key.clone();
        handles.push(std::thread::spawn(move || {
            let mut led = IdempotencyLedger::open(&p).unwrap();
            b.wait();
            led.admit(&k).unwrap()
        }));
    }
    let firsts = handles
        .into_iter()
        .map(|h| h.join().unwrap())
        .filter(|r| *r == Admission::First)
        .count();
    let mut reloaded = IdempotencyLedger::open(&path).unwrap();
    let durable_keys = reloaded.len();
    assert_eq!(reloaded.admit(&key).unwrap(), Admission::Duplicate);
    report(
        "V-A5b",
        if firsts > 1 { "RACE-OBSERVED-DEBT-RECORDED" } else { "CONTRACT-HELD" },
        &format!("first_outcomes={firsts}/2 durable_unique_keys={durable_keys}; no locking between independent admitters (no production multi-writer exists yet)"),
    );
}
